        // मात्रा (Quantity) बदलने का फंक्शन
        function changeQty(id, delta) {
            const current = cart[id] || 0;
            const next = Math.max(0, current + delta);
            if (next === 0) {
                delete cart[id];
            } else {
                cart[id] = next;
            }

            // कार्ड UI अपडेट करें
            const qtyEl = document.getElementById("qty-" + id);
            const cardEl = document.getElementById("card-" + id);
            if (qtyEl) {
                qtyEl.textContent = next;
                qtyEl.classList.toggle("active", next > 0);
            }
            if (cardEl) {
                cardEl.classList.toggle("selected", next > 0);
            }

            updateSummary();
        }

        // कैटेगरी फ़िल्टर (All / Veg / Non-Veg)
        function filterMenu(category, btn) {
            document.querySelectorAll(".filter-btn").forEach(b => b.classList.remove("active"));
            btn.classList.add("active");

            document.querySelectorAll(".menu-card").forEach(card => {
                const type = card.getAttribute("data-type");
                if (category === "all" || type === category) {
                    card.classList.remove("hidden-item");
                } else {
                    card.classList.add("hidden-item");
                }
            });
        }

        // कस्टम नोट में क्विक टैग जोड़ना
        function addQuickTag(text) {
            const box = document.getElementById("custom-note");
            const val = box.value.trim();
            if (!val.includes(text)) {
                box.value = val ? val + ", " + text : text;
                updateSummary();
                showToast('"' + text + '" जोड़ा गया', "✍️");
            }
        }

        // कार्ट सारांश और स्टिकी बार अपडेट करना
        function updateSummary() {
            let totalQty = 0;
            let totalRupees = 0;
            const listEl = document.getElementById("receipt-list");
            listEl.innerHTML = "";

            Object.keys(ITEMS).forEach(id => {
                const qty = cart[id] || 0;
                if (qty > 0) {
                    const item = ITEMS[id];
                    const sub = qty * item.price;
                    totalQty += qty;
                    totalRupees += sub;

                    const li = document.createElement("li");
                    li.innerHTML = `<span><strong>${item.nameHi}</strong> (${qty} ${item.unit})</span><span><strong>₹${sub}</strong></span>`;
                    listEl.appendChild(li);
                }
            });

            const customText = document.getElementById("custom-note").value.trim();
            if (customText) {
                const li = document.createElement("li");
                li.style.color = "#9a3412";
                li.style.background = "#ffedd5";
                li.style.padding = "5px 8px";
                li.style.borderRadius = "8px";
                li.style.marginTop = "4px";
                li.textContent = "✍️ अन्य / कस्टम: " + customText;
                listEl.appendChild(li);
            }

            const hasItems = totalQty > 0 || customText.length > 0;
            document.getElementById("sticky-bar").classList.toggle("visible", hasItems);
            document.getElementById("receipt-box").classList.toggle("visible", hasItems);

            document.getElementById("sticky-count").textContent = totalQty;
            document.getElementById("sticky-label").textContent = totalQty > 0 ? `${totalQty} आइटम चुने गए` : "कस्टम ऑर्डर लिखा गया";
            document.getElementById("sticky-price").textContent = "₹" + totalRupees;
            document.getElementById("receipt-total").textContent = "₹" + totalRupees;
        }

        // पूरा ऑर्डर रीसेट करना
        function resetOrder() {
            Object.keys(cart).forEach(id => {
                delete cart[id];
                const qtyEl = document.getElementById("qty-" + id);
                const cardEl = document.getElementById("card-" + id);
                if (qtyEl) { qtyEl.textContent = "0"; qtyEl.classList.remove("active"); }
                if (cardEl) { cardEl.classList.remove("selected"); }
            });
            document.getElementById("custom-note").value = "";
            updateSummary();
            showToast("ऑर्डर साफ़ कर दिया गया है", "🗑️");
        }

        // WhatsApp पर ऑर्डर भेजना
        function submitWhatsAppOrder() {
            const customText = document.getElementById("custom-note").value.trim();
            const hasMenuItems = Object.keys(cart).length > 0;

            if (!hasMenuItems && !customText) {
                showToast("कृपया पहले मेन्यू से खाना चुनें या 'अन्य' बॉक्स में लिखें!", "🍲");
                window.scrollTo({ top: 0, behavior: "smooth" });
                return;
            }

            const name = document.getElementById("cust-name").value.trim();
            const phone = document.getElementById("cust-phone").value.trim();
            const address = document.getElementById("cust-address").value.trim();

            if (!name) {
                showToast("कृपया अपना नाम लिखें", "👤");
                document.getElementById("cust-name").focus();
                return;
            }
            if (!phone || phone.length < 10) {
                showToast("कृपया सही 10 अंकों का मोबाइल नंबर लिखें", "📱");
                document.getElementById("cust-phone").focus();
                return;
            }
            if (!address) {
                showToast("कृपया डिलीवरी का पता या टेबल नंबर लिखें", "📍");
                document.getElementById("cust-address").focus();
                return;
            }

            let msg = `*🍲 नया फूड ऑर्डर - ${KITCHEN_TITLE}*\n`;
            msg += `----------------------------------\n`;
            msg += `*👤 ग्राहक का नाम:* ${name}\n`;
            msg += `*📞 मोबाइल:* ${phone}\n`;
            msg += `*📍 पता / टेबल:* ${address}\n`;
            msg += `----------------------------------\n`;
            msg += `*📋 ऑर्डर लिस्ट:*\n`;

            let total = 0;
            let count = 1;
            Object.keys(ITEMS).forEach(id => {
                const qty = cart[id] || 0;
                if (qty > 0) {
                    const item = ITEMS[id];
                    const sub = qty * item.price;
                    total += sub;
                    msg += `${count}. ${item.nameHi} (${item.nameEn}) × *${qty} ${item.unit}* = ₹${sub}\n`;
                    count++;
                }
            });

            if (customText) {
                msg += `\n*✍️ अन्य / अपनी पसंद का खाना (Custom Order):*\n"${customText}"\n`;
            }

            msg += `----------------------------------\n`;
            if (total > 0) {
                msg += `*💰 कुल अनुमानित बिल: ₹${total}*\n`;
            }
            if (customText) {
                msg += `_(नोट: कस्टम आइटम का रेट किचन द्वारा कन्फर्म किया जाएगा)_\n`;
            }
            msg += `\nकृपया मेरा ऑर्डर कन्फर्म करें। धन्यवाद! 🙏`;

            const url = `https://wa.me/${KITCHEN_WHATSAPP_NUMBER}?text=${encodeURIComponent(msg)}`;
            showToast("WhatsApp खुल रहा है...", "✅");
            window.open(url, "_blank");
        }

        // QR कोड मॉडल खोलना (ऑनलाइन API + ऑफलाइन SVG फॉलबैक के साथ)
        function openQRModal() {
            const pageUrl = window.location.href;
            document.getElementById("qr-link-text").textContent = pageUrl;

            const container = document.getElementById("qr-container");
            const encodedUrl = encodeURIComponent(pageUrl);
            const qrApiSrc = `https://api.qrserver.com/v1/create-qr-code/?size=190x190&color=431407&data=${encodedUrl}`;

            container.innerHTML = `<img src="${qrApiSrc}" alt="Scan to Order QR Code" width="190" height="190" style="display:block; border-radius:8px;" onerror="renderOfflineQRFallback(this.parentElement)">`;
            document.getElementById("qr-modal").classList.add("open");
        }

        function copyPageUrl() {
            const temp = document.createElement("input");
            temp.value = window.location.href;
            document.body.appendChild(temp);
            temp.select();
            document.execCommand("copy");
            document.body.removeChild(temp);
            showToast("वेबसाइट लिंक कॉपी हो गया है!", "🔗");
        }

        let toastTimer;
        function showToast(message, icon) {
            const toast = document.getElementById("toast");
            document.getElementById("toast-msg").textContent = message;
            document.getElementById("toast-icon").textContent = icon || "ℹ️";
            toast.classList.add("show");
            clearTimeout(toastTimer);
            toastTimer = setTimeout(() => toast.classList.remove("show"), 3200);
        }
